curl -X POST http://127.0.0.1:8000/api/voice-detection   -H "Content-Type: application/json"   -d '{
    "language": "English",
    "audioFormat": "wav",
    "audioBase64": "<BASE64_ENCODED_AUDIO_STRING>"
  }'
import requests
import base64

with open("voice_sample.wav", "rb") as f:
    audio_b64 = base64.b64encode(f.read()).decode("utf-8")

payload = {
    "language": "English",
    "audioFormat": "wav",
    "audioBase64": audio_b64
}

response = requests.post("http://127.0.0.1:8000/api/voice-detection", json=payload)
print(response.json())
